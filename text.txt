    function generateCalendar() {
        const cropKey = document.getElementById('cropSelect').value;
        const plan = cropPlans[cropKey];
        const resultDiv = document.getElementById('calendarResult');

        let tasksHtml = plan.tasks.map(t => `
            <div class="task-item">
                <div>
                    <span class="task-date">Week ${t.week}:</span> 
                    <strong>${t.task}</strong>
                    <p class="text-sm" style="color: #555; margin-top: 2px;">${t.detail}</p>
                </div>
            </div>
        `).join('');

            <div class="card">
                <span class="badge">Farm Plan: ${plan.name}</span>
                <p style="margin-bottom: 8px;"><strong>Fertilizer Strategy:</strong> ${plan.fertilizer}</p>

            </div>
        `;

    // 3. Image Preview & Diagnosis
    function previewImage(event) {
        const file = event.target.files[0];
        if (file) {
            const reader = new FileReader();
            reader.onload = function(e) {
                const img = document.getElementById('imagePreview');
                img.src = e.target.result;
                img.classList.remove('hidden');
                document.getElementById('diagnoseBtn').classList.remove('hidden');
                document.getElementById('diagnosisResult').classList.remove('show');
            }
            reader.readAsDataURL(file);
        }
    }
